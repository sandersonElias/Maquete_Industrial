Código & Automação

Como os caminhões se movem e como o trem funciona

ferrovia_firmware.ino 
// Controle do trem via Bluetooth
void executarComando(String cmd) {
  if (cmd == "F") {
    servoMotor.write(180); // Frente
  }
  if (cmd == "B") {
    servoMotor.write(0);   // Ré
  }
  if (cmd == "S") {
    servoMotor.write(90);  // Parado
  }
}
